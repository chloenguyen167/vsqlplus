3. CAI TIEN 1 - Viet lai tat dinh: dummy SQL -> SQL tren bang goc (khong goi LLM)
==============================================================================
Dummy SQL tren view : SELECT superhero_name FROM v_superhero WHERE publisher_name = 'Marvel Comics' AND hair_colour = 'Blond'

Sau khi mo rong view, prune=False (7 LEFT JOIN):
  WITH v_superhero AS (SELECT superhero.id AS id, superhero.superhero_name AS superhero_name, superhero.full_name AS full_name, superhero.gender_id AS gender_id, superhero.eye_colour_id AS eye_colour_id, superhero.hair_colour_id AS hair_colour_id, superhero.skin_colour_id AS skin_colour_id, superhero.race_id AS race_id, superhero.publisher_id AS publisher_id, superhero.alignment_id AS alignment_id, superhero.height_cm AS height_cm, superhero.weight_kg AS weight_kg, alignment.alignment AS alignment, publisher.publisher_name AS publisher_name, race.race AS race, skin_colour.colour AS skin_colour, hair_colour.colour AS hair_colour, eye_colour.colour AS eye_colour, gender.gender AS gender FROM superhero LEFT JOIN alignment AS alignment ON superhero.alignment_id = alignment.id LEFT JOIN publisher AS publisher ON superhero.publisher_id = publisher.id LEFT JOIN race AS race ON superhero.race_id = race.id LEFT JOIN colour AS skin_colour ON superhero.skin_colour_id = skin_colour.id LEFT JOIN colour AS hair_colour ON superhero.hair_colour_id = hair_colour.id LEFT JOIN colour AS eye_colour ON superhero.eye_colour_id = eye_colour.id LEFT JOIN gender AS gender ON superhero.gender_id = gender.id) SELECT superhero_name FROM v_superhero WHERE publisher_name = 'Marvel Comics' AND hair_colour = 'Blond'

Sau khi mo rong view, prune=True (2 LEFT JOIN):
  WITH v_superhero AS (SELECT superhero.id AS id, superhero.superhero_name AS superhero_name, superhero.full_name AS full_name, superhero.gender_id AS gender_id, superhero.eye_colour_id AS eye_colour_id, superhero.hair_colour_id AS hair_colour_id, superhero.skin_colour_id AS skin_colour_id, superhero.race_id AS race_id, superhero.publisher_id AS publisher_id, superhero.alignment_id AS alignment_id, superhero.height_cm AS height_cm, superhero.weight_kg AS weight_kg, publisher.publisher_name AS publisher_name, hair_colour.colour AS hair_colour FROM superhero LEFT JOIN publisher AS publisher ON superhero.publisher_id = publisher.id LEFT JOIN colour AS hair_colour ON superhero.hair_colour_id = hair_colour.id) SELECT superhero_name FROM v_superhero WHERE publisher_name = 'Marvel Comics' AND hair_colour = 'Blond'

Doi chung voi view that cua SQLite: 24/24 phep thu cho ket qua giong het (12 truy van x {co, khong} cat JOIN thua)
